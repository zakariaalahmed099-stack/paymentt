-cpwd                     # عرض المسار الحالي

cd ..                   # الرجوع إلى المجلد الأب

cd Desktop              # الدخول إلى مجلد Desktop

ls                      # عرض محتويات المجلد الحالي

mkdir 1 2 3 4 5 6 7 8 9 # إنشاء عدة مجلدات دفعة واحدة

rm -d 5 6 7 8 9         # حذف المجلدات الفارغة من 5 إلى 9

touch 1.txt 2.txt 3.txt 4.txt   # إنشاء ملفات نصية فارغة

ls -l                   # عرض الملفات مع التفاصيل (الصلاحيات، الحجم...)
ls -a                   # عرض جميع الملفات بما فيها المخفية

touch .txt              # إنشاء ملف مخفي اسمه .txt

ls                      # عرض محتويات المجلد

mkdir "1 1"             # إنشاء مجلد يحتوي اسمه على فراغ

clear                   # مسح شاشة التيرمنل

mkdir "1 1" "1m 2"      # إنشاء مجلدين بأسماء تحتوي على فراغات

mkdir "1 1" 1           # إنشاء مجلد اسمه "1 1" وآخر اسمه 1

rm -d "1m 2"            # حذف مجلد فارغ اسمه يحتوي على فراغ

mv "11" "1 2"           # إعادة تسمية 11 إلى 1 2

mv "1 1" "1 2"          # تغيير اسم مجلد يحتوي فراغ

mv "1 2" h              # نقل المجلد 1 2 إلى داخل المجلد h

rm -d 4                 # حذف المجلد 4 إذا كان فارغًا

rm -r 3                 # حذف المجلد 3 مع كل محتوياته

df -h                   # عرض مساحة القرص بصيغة مقروءة

echo >4.txt             # تفريغ محتوى الملف 4.txt أو إنشاؤه

echo "hello" >4.txt     # كتابة hello داخل الملف مع حذف المحتوى السابق

echo "hellof" >>4.txt   # إضافة hellof إلى نهاية الملف دون حذف السابق




head 1.txt               # عرض أول 10 أسطر من الملف 1.txt

tail 1.txt               # عرض آخر 10 أسطر من الملف 1.txt

grep "hi" 1.txt          # البحث عن الكلمة "hi" داخل الملف 1.txt

grep "hi" *.txt          # البحث عن "hi" داخل جميع الملفات التي تنتهي بـ .txt

cp 1.txt 1               # نسخ الملف 1.txt إلى المجلد 1

grep -r "hi" 1           # البحث عن "hi" داخل المجلد 1 وكل محتوياته

cp -r 11 1               # نسخ المجلد 11 مع كل محتوياته إلى المجلد 1

top                      # عرض العمليات الجارية واستهلاك المعالج والذاكرة (مباشر)

free -h                  # عرض الذاكرة المستعملة والمتوفرة بصيغة مقروءة

help                     # عرض المساعدة العامة لسطر الأوامر

exit                     # الخروج من التيرمنل أو الجلسة الحالية

gzip 1.txt               # ضغط الملف 1.txt (ويُحذف الملف الأصلي)

gunzip 1.txt.gz          # فك ضغط الملف 1.txt.gz

gzip -k 1.txt            # ضغط الملف مع الاحتفاظ بالملف الأصلي

gunzip -k 1.txt.gz       # فك الضغط مع الاحتفاظ بالملف المضغوط

tar -cvf 111.tar 1       # إنشاء أرشيف tar باسم 111.tar للمجلد 1

tar -czvf 112.tar.gz 1   # إنشاء أرشيف tar مضغوط gzip للمجلد 1

tar -xzvf 112.tar.gz     # فك ضغط واستخراج ملف 112.tar.gz

tar -xvf 112.tar         # استخراج محتوى الملف 112.tar

tar -tvf 112.tar         # عرض محتويات الأرشيف 112.tar دون فكّه

tar -tvf 112.tar.gz      # عرض محتويات الأرشيف المضغوط 112.tar.gz

history                  # عرض جميع الأوامر التي تم تنفيذها سابقًا

ping 8.8.8.8             # اختبار الاتصال بالشبكة مع Google DNS
